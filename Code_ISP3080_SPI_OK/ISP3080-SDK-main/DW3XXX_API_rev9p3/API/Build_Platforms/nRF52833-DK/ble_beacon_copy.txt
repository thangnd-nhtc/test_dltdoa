#include <string.h>
#include "ble_beacon.h"

#include "nrf_sdh_ble.h"
#include "ble_advdata.h"
#include "app_error.h"

#define APP_ADV_DATA_LENGTH   0x15
#define APP_DEVICE_TYPE       0x02    /* Beacon */
#define BEACON_INFO_LEN       0x17

static ble_gap_adv_params_t  m_adv_params;
static uint8_t               m_adv_handle = BLE_GAP_ADV_SET_HANDLE_NOT_SET;
static uint8_t               m_enc_advdata[BLE_GAP_ADV_SET_DATA_SIZE_MAX];

static ble_gap_adv_data_t m_adv_data = {
    .adv_data = { .p_data = m_enc_advdata, .len = BLE_GAP_ADV_SET_DATA_SIZE_MAX },
    .scan_rsp_data = { .p_data = NULL, .len = 0 }
};

/* M?ng payload “iBeacon-like” */
static uint8_t m_beacon_info[BEACON_INFO_LEN] = {
    APP_DEVICE_TYPE,          /* 0  */
    APP_ADV_DATA_LENGTH,      /* 1  */
    BLE_BEACON_UUID_128,      /* 2..17 */
    BLE_BEACON_MAJOR,         /* 18..19 */
    BLE_BEACON_MINOR,         /* 20..21 */
    BLE_BEACON_TX_PWR         /* 22 */
};

void ble_beacon_init(void)
{
    uint32_t err;
    ble_advdata_t advdata;
    ble_advdata_manuf_data_t mfg;

    mfg.company_identifier   = BLE_BEACON_COMPANY_ID;
    mfg.data.p_data          = m_beacon_info;
    mfg.data.size            = sizeof(m_beacon_info);

    memset(&advdata, 0, sizeof(advdata));
    advdata.name_type             = BLE_ADVDATA_NO_NAME;
    advdata.flags                 = BLE_GAP_ADV_FLAG_BR_EDR_NOT_SUPPORTED;
    advdata.p_manuf_specific_data = &mfg;

    memset(&m_adv_params, 0, sizeof(m_adv_params));
    m_adv_params.properties.type  = BLE_GAP_ADV_TYPE_NONCONNECTABLE_NONSCANNABLE_UNDIRECTED;
    m_adv_params.p_peer_addr      = NULL;
    m_adv_params.filter_policy    = BLE_GAP_ADV_FP_ANY;
    m_adv_params.interval         = BLE_BEACON_ADV_INTERVAL_UNITS;
    m_adv_params.duration         = 0;

    m_adv_data.adv_data.len = BLE_GAP_ADV_SET_DATA_SIZE_MAX;
    err = ble_advdata_encode(&advdata, m_adv_data.adv_data.p_data, &m_adv_data.adv_data.len);
    APP_ERROR_CHECK(err);

    err = sd_ble_gap_adv_set_configure(&m_adv_handle, &m_adv_data, &m_adv_params);
    APP_ERROR_CHECK(err);
}

void ble_beacon_start(void)
{
    uint32_t err = sd_ble_gap_adv_start(m_adv_handle, BLE_BEACON_CONN_CFG_TAG);
    APP_ERROR_CHECK(err);
}
